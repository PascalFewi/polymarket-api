Get Orderbook Data

Retrieve historical orderbook snapshots for any market

// Fetch orderbook data for a specific market
const getOrderbookData = async () => {
  const response = await fetch('https://api.deltaodds.io/v1/orderbook', {
    method: 'GET',
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY',
      'Content-Type': 'application/json'
    },
    params: {
      market_id: '0x1234...',
      start_time: '2024-01-01T00:00:00Z',
      end_time: '2024-01-31T23:59:59Z',
      interval: '1m' // 1 minute snapshots
    }
  });

  const data = await response.json();
  return data;
};

Python Example

Perfect for data science and quantitative research workflows

import requests
import pandas as pd

# Fetch orderbook data
def get_orderbook_data(market_id, start_time, end_time):
    url = 'https://api.deltaodds.io/v1/orderbook'
    headers = {
        'Authorization': 'Bearer YOUR_API_KEY',
        'Content-Type': 'application/json'
    }
    params = {
        'market_id': market_id,
        'start_time': start_time,
        'end_time': end_time,
        'interval': '5m'
    }

    response = requests.get(url, headers=headers, params=params)
    data = response.json()

    # Convert to DataFrame for analysis
    df = pd.DataFrame(data['snapshots'])
    return df

# Analyze probability changes over time
df = get_orderbook_data('0x1234...', '2024-01-01', '2024-01-31')
df['probability'] = df['mid_price']
df['probability_change'] = df['probability'].diff()
